<!DOCTYPE 
        html>
      
      <
        html>
      
      <
        head>
      
      <
        title>
      Ma Bibliothèque BDBOOK
      </
        title>
      
      <
        style>
      
        body { font-family: sans-serif; padding: 20px; }
        input { padding: 10px; width: 300px; }
        #results { margin-top: 20px; }
      </
        style>
      
      </
        head>
      
      <
        body>
      
      <
        h1>
      Catalogue BDBOOK
      </
        h1>
      
      <
        input 
        type=
        "text" 
        id=
        "search" 
        placeholder=
        "Rechercher par sujet ou titre...">
      
      <
        div 
        id=
        "results">
      
      </
        div>
      

      
      <
        script>
      
      
        fetch(
        'catalogue.json')
            .
        then(
        
          res =>
         res.
        json())
            .
        then(
        
          data =>
         {
        document.
        getElementById(
        'search').
        addEventListener(
        'input', 
        (
          e) =>
         {
        const val = e.
        target.
        value.
        toLowerCase();
        const filtered = data.
        filter(
        
          book =>
         
                        book.
        titre.
        toLowerCase().
        includes(val) || 
                        book.
        sujet.
        toLowerCase().
        includes(val)
                    );
        render(filtered);
                });
        render(data);
            });

        
        function 
        render(
        books) {
        const container = 
        document.
        getElementById(
        'results');
            container.
        innerHTML = books.
        map(
        
          b =>
         
        `<p>
          ${b.titre} - <b>
          ${b.sujet}</b></p>`
        ).
        join(
        '');
        }
      
      </
        script>
      
      </
        body>
      
      </
        html>
